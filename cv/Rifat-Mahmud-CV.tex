\section{Projects}

\begin{tabular*}{\textwidth}{@{}l@{\extracolsep{\fill}}r@{}}
    \textbf{Explainable Hybrid Ensemble Diagnosis of Five Lung Conditions} & ULAB thesis, 2026 \\
\end{tabular*}
\begin{itemize}
    \item Five-class chest X-ray system (bacterial pneumonia, viral pneumonia, COVID-19, tuberculosis, normal) that segments both lungs, crops the film, and refuses a label when the upload fails the lung check
    \item Fine-tuned DenseNet201, ResNet50, VGG16, and EfficientNet-B3 on 35,686 lung-cropped images with one shared two-phase schedule
    \item Soft-averages the four models for the pneumonia classes and COVID-19; normal versus tuberculosis uses EfficientNet-B3, a horizontal flip, and a 0.6 threshold
    \item 89.23\% accuracy and 91.67\% macro-F1 on 7,651 held-out images. Grad-CAM++, LIME, and SHAP stay inside the lung mask. Research prototype
    \item \textit{Tech Stack: Python, PyTorch, OpenCV, PSPNet, CLAHE, Grad-CAM++, LIME, SHAP, Flask}
\end{itemize}
